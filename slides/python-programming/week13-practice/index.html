class Student:

def __init__(self, name, student_id, year):

self.name = name

self.student_id = student_id

self.year = year

self.scores = []

def add_score(self, score):

self.scores.append(score)

def average(self):

return sum(self.scores) / len(self.scores)

def info(self):

print(f"{self.name} ({self.student_id}), year {self.year}")

class GradStudent(Student):

def __init__(self, name, student_id, year, lab):

super().__init__(name, student_id, year)

self.lab = lab

self.papers = 0

def info(self):

super().info()

print(f"Graduate student in {self.lab}")

def write_paper(self):

self.papers += 1

class Rectangle:

def __init__(self, width, height):

self.width = width

self.height = height

def area(self):

return self.width * self.height

def info(self):

print(f"Rectangle {self.width} x {self.height}")

class Square(Rectangle):

def __init__(self, side):

super().__init__(side, side)

def info(self):

print(f"Square {self.width}")

shapes = [Rectangle(3, 4), Square(5), Rectangle(2, 6)]

for s in shapes:

s.info()

print(s.area())

class Account:

def __init__(self, owner):

"""Keep the owner; the balance starts at 0."""

self.owner = owner

self.balance = 0

def deposit(self, amount):

"""Add amount to the balance."""

self.balance += amount

def withdraw(self, amount):

"""Take amount away, or print Not enough money."""

if amount > self.balance:

print("Not enough money.")

else:

self.balance -= amount

def show(self):

"""Print the owner and the balance."""

print(f"{self.owner}: {self.balance} won")
