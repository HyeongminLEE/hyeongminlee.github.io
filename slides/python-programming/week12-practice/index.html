resistor.py

class Resistor:

def __init__(self, ohms):

self.ohms = ohms

def current(self, voltage):

return voltage / self.ohms * 1000

def power(self, voltage):

return voltage * voltage / self.ohms * 1000

resistors = [Resistor(100), Resistor(220), Resistor(1000)]

voltage = float(input("Voltage: "))

for r in resistors:

current = r.current(voltage)

power = r.power(voltage)

print(f"{r.ohms} ohms: {current:.1f} mA, {power:.1f} mW")

Bank account: the class

class Account:

def __init__(self, owner):

"""Keep the owner; the balance starts at 0."""

self.owner = owner

self.balance = 0

1

def deposit(self, amount):

"""Add amount to the balance."""

self.balance += amount

2

def withdraw(self, amount):

"""Take amount away, or print Not enough money."""

if amount > self.balance:

print("Not enough money.")

else:

self.balance -= amount

3

def show(self):

"""Print the owner and the balance."""

print(f"{self.owner}: {self.balance} won")

Power strip

week12_problem1.py 2 points

A power strip has three devices. Each keeps its name, its watts, and whether it is on (all start off). The class Device is started for you, and the program below it is given: write the body of each method.

in the file a dashed box is a body to write

class Device:

def __init__(self, name, watts):

keep name and watts; on starts as False

def switch(self):

turn on if off, and off if on

def power(self):

return watts if on, and 0 if off

tv = Device("TV", 120)

heater = Device("Heater", 2000)

lamp = Device("Lamp", 30)

devices = [tv, heater, lamp]

while True:

name = input("Switch: ")

if name == "":

break

for d in devices:

if d.name == name:

d.switch()

total = 0

for d in devices:

total += d.power()

print(f"Total: {total} W")
