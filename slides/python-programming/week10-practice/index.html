the code

import random

secret = random.randint(1, 100)

tries = 0

while True:

guess = int(input("Guess: "))

tries += 1

if guess < secret:

print("Up")

elif guess > secret:

print("Down")

else:

print(f"Correct! Tries: {tries}")

break
